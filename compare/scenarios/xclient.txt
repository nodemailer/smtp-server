# XCLIENT and XFORWARD (http://www.postfix.org/XCLIENT_README.html, XFORWARD_README.html)
EHLO proxy.example.com
XCLIENT
XCLIENT FOO=bar
XCLIENT ADDR=not-an-ip
XCLIENT NAME=bad!name
XFORWARD
XFORWARD FOO=bar
XFORWARD ADDR=192.0.2.1 NAME=client.example.com PORT=1234 PROTO=ESMTP HELO=helo.example.com IDENT=abc SOURCE=REMOTE
XFORWARD ADDR=[UNAVAILABLE] NAME=[TEMPUNAVAIL]
MAIL FROM:<sender@example.com>
XFORWARD ADDR=192.0.2.2
XCLIENT ADDR=192.0.2.2
RSET
